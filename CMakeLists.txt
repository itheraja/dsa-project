cmake_minimum_required(VERSION 3.20)
project(SolanaScope VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(POLICY CMP0135)
 cmake_policy(SET CMP0135 NEW)
endif()
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_Declare(httplib URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.3.tar.gz)
set(HTTPLIB_REQUIRE_OPENSSL OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json httplib)
add_library(scope core/src/Graph.cpp core/src/Algorithms.cpp core/src/Engine.cpp)
target_include_directories(scope PUBLIC core/include)
target_link_libraries(scope PUBLIC nlohmann_json::nlohmann_json)
target_compile_definitions(scope PUBLIC SCOPE_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
target_compile_options(scope PRIVATE -Wall -Wextra -Wconversion)
add_executable(solanascope backend/main.cpp)
target_link_libraries(solanascope PRIVATE scope httplib::httplib)
add_executable(scope_benchmark benchmarks/main.cpp)
target_link_libraries(scope_benchmark PRIVATE scope)
include(CTest)
if(BUILD_TESTING)
 FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz)
 FetchContent_MakeAvailable(googletest)
 add_executable(scope_tests tests/core_tests.cpp)
 target_link_libraries(scope_tests PRIVATE scope GTest::gtest_main)
 include(GoogleTest)
 gtest_discover_tests(scope_tests)
endif()
if(BUILD_TESTING)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME API.Integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/api_integration.py $<TARGET_FILE:solanascope>)
endif()
